ISA

0000. ingnore
001x. add one to reg
	0010. A
	0011. B
01xx. move stack to reg
	01x1. reg b
	011x. stack b
10xx. move reg to stack
	10x1. reg b
	101x. stack b 
1100. add reg b to reg a
1101. sub reb a from reg b
	(final output of alu oprations ends up in reg a)
1110. compare if reg a == reg b (if not then signal to ignore everything until 1111 is read)
1111. end of if statement

ASM

nop   - do nothing

adda  - add one to register A
addb  - add one to register B

mvsr  - move register A to the stack
mvrs  - move the top value of the stack to register A

add   - add register B to register A (A = A + B)
sub   - subtract register B from register A (A = A - B)

com   - compare register A and register B
        com 1 = A == B
        com 0 = A != B
        if false, ignore instructions until ecom

ecom  - end a conditional block

#     - comment line
